Formula Transparency

delegable_admin_hours = max(physician_admin_hours − 4, 0)
physician_admin_hours_recovered = delegable_admin_hours × 0.7
recovered_hours_per_week_per_va = min(physician_admin_hours_recovered, 40)
clinical_equivalent_hours_per_va = recovered_hours_per_week_per_va × 0.3
patients_per_admin_hour = patients_seen_per_week ÷ physician_admin_hours
raw_extra_patients_per_week_per_va = patients_per_admin_hour × clinical_equivalent_hours_per_va
extra_patients_per_week_per_va = min(raw_extra_patients_per_week_per_va, patients_seen_per_week × 0.2)
va_throughput_multiplier = (1 − 0.45^va_count) ÷ (1 − 0.45)
total_extra_patients_per_week = floor(extra_patients_per_week_per_va × va_throughput_multiplier)
extra_patients_per_year = total_extra_patients_per_week × 48
additional_revenue = extra_patients_per_year × revenue_per_visit
additional_overhead = extra_patients_per_year × overhead_per_visit
additional_margin = additional_revenue − additional_overhead (already includes exponential scaling)
annual_va_cost = $19,200 × va_count
net_benefit = additional_margin − annual_va_cost
in_house_ma_total = (salary + benefits + training) × va_count
va_advantage_vs_ma = net_benefit − ((benefits + training) × va_count)
