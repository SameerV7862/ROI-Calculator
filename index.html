Formula Transparency

recovered_hrs = admin_hours × 0.7
annual_recovered_hrs = recovered_hrs × 48
effective_visit_cycle_minutes = visit_time_minutes × 2.5
extra_patients_per_week = floor(recovered_hrs × 60 ÷ effective_visit_cycle_minutes)
extra_patients_per_year = extra_patients_per_week × 48
additional_revenue = extra_patients_per_year × revenue_per_visit
additional_overhead = extra_patients_per_year × overhead_per_visit
additional_margin = additional_revenue − additional_overhead
annual_va_cost = $19,200 × va_count
net_benefit = (additional_margin × va_count) − annual_va_cost
in_house_ma_total = (salary + benefits + training) × va_count
va_advantage_vs_ma = net_benefit − ((benefits + training) × va_count)
